type appFlowManifest struct {
	Task, OtherTask, Project, Model, Mode string
	Provider, Transport                   string
	OutputBytes, ToolBytes, Turns         int
}

func appFlowCPU() (float64, int64) {
	var r syscall.Rusage
	_ = syscall.Getrusage(syscall.RUSAGE_SELF, &r)
	return float64(r.Utime.Sec+r.Stime.Sec)*1000 + float64(r.Utime.Usec+r.Stime.Usec)/1000, r.Maxrss
}
func appFlowJSON(t *testing.T, path string, value any) {
	t.Helper()
	b, e := json.MarshalIndent(value, "", "  ")
	require.NoError(t, e)
	require.NoError(t, os.WriteFile(path, b, 0600))
}

// Full allocation sampling is diagnostic only. Set once before test setup so
// profiles have a consistent rate; never compare these runs with normal timings.
func init() {
	if os.Getenv("OV_FLOW_DIAGNOSTICS") == "allocations" {
		runtime.MemProfileRate = 1
	}
}

func TestConcurrentFlow(t *testing.T) {
	dir := os.Getenv("OV_FLOW_DIR")
	if dir == "" {
		t.Skip("opt-in app flow")
	}
	markPhase, finishMemory := appFlowMemoryPhases(t, dir)
	defer finishMemory()
	markPhase("startup")
	ctx := context.Background()
	diagnosticMode := os.Getenv("OV_FLOW_DIAGNOSTICS")
	diagnosticsEnabled := diagnosticMode == "1" || diagnosticMode == "trace"
	allocationDiagnostic := diagnosticMode == "allocations"
	if allocationDiagnostic {
		appFlowJSON(t, filepath.Join(dir, "allocation-diagnostic.json"), map[string]any{"mem_profile_rate": runtime.MemProfileRate, "forced_gc_at_snapshots": true, "performance_comparison": false})
	}
	conns, err := database.NewReadWrite(filepath.Join(dir, "app.db"))
	require.NoError(t, err)
	defer conns.Close()
	unregister := sqlite.RegisterDedicatedWriter(conns.Reader, conns.Writer)
	defer unregister()
	broadcaster := events.NewBroadcaster()
	stores := sqlite.NewStores(conns.Reader, broadcaster)
	env := wireTestHandler(t, stores, testHandlerOptions{broadcaster: broadcaster})
	env.LLMSvc.SetLLMCaller(nil)
	hub := events.NewExecutionStreamHub()
	env.LLMSvc.SetExecutionStreamHub(hub)
	env.Handler.SetExecutionStreamHub(hub)
	env.LLMSvc.SetOpenAIAsyncToolCallStore(stores.OpenAIAsyncToolCalls)
	// Constant-size per-execution observer: no text retained, no per-delta sample allocation.
	type delivery struct {
		ID                                      string
		Deltas, Persisted                       int
		FirstMS, MaxDeltaGapMS, MaxPersistGapMS float64
	}
	var deliveries []delivery
	var deliveryMu sync.Mutex
	var observers sync.WaitGroup
	var subscriptions sync.Map
	subscribe := func(id string) {
		if _, loaded := subscriptions.LoadOrStore(id, true); loaded {
			return
		}
		sub, unsub, e := hub.Subscribe(id)
		if e != nil {
			t.Error(e)
			return
		}
		observers.Add(1)
		go func(id string, sub events.ExecutionStreamSubscriber, unsub func()) {
			defer observers.Done()
			defer unsub()
			start := time.Now()
			lastDelta, lastPersist := time.Time{}, time.Time{}
			d := delivery{ID: id}
			for event := range sub {
				now := time.Now()
				switch event.Type {
				case events.ExecutionStreamDelta:
					if d.Deltas == 0 {
						d.FirstMS = float64(now.Sub(start).Microseconds()) / 1000
					}
					d.Deltas++
					if !lastDelta.IsZero() {
						gap := float64(now.Sub(lastDelta).Microseconds()) / 1000
						if diagnosticsEnabled && gap >= 10 {
							trace.Logf(ctx, "live-gap", "exec=%s gap_ms=%.3f", id, gap)
						}
						if gap > d.MaxDeltaGapMS {
							d.MaxDeltaGapMS = gap
						}
					}
					lastDelta = now
				case events.ExecutionStreamPersisted:
					d.Persisted++
					if !lastPersist.IsZero() {
						gap := float64(now.Sub(lastPersist).Microseconds()) / 1000
						if diagnosticsEnabled && gap >= 510 {
							trace.Logf(ctx, "persist-gap", "exec=%s gap_ms=%.3f", id, gap)
						}
						if gap > d.MaxPersistGapMS {
							d.MaxPersistGapMS = gap
						}
					}
					lastPersist = now
				}
			}
			deliveryMu.Lock()
			deliveries = append(deliveries, d)
			deliveryMu.Unlock()
		}(id, sub, unsub)
	}
	eventSub, e := broadcaster.Subscribe()
	require.NoError(t, e)
	eventDone := make(chan struct{})
	go func() {
		defer close(eventDone)
		for event := range eventSub {
			if event.ExecID != "" {
				subscribe(event.ExecID)
			}
		}
	}()
	var manifest appFlowManifest
	if os.Getenv("OV_FLOW_SERVE") == "1" {
		raw, e := os.ReadFile(filepath.Join(dir, "manifest.json"))
		require.NoError(t, e)
		require.NoError(t, json.Unmarshal(raw, &manifest))
		server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			if !static.ServeAsset(w, r) {
				env.Echo.ServeHTTP(w, r)
			}
		}))
		defer server.Close()
		require.NoError(t, os.WriteFile(filepath.Join(dir, "app-url"), []byte(server.URL), 0600))
		for {
			if _, e := os.Stat(filepath.Join(dir, "stop-server")); e == nil {
				return
			}
			time.Sleep(100 * time.Millisecond)
		}
	}
	modelRaw, e := os.ReadFile(filepath.Join(dir, "model-url"))
	require.NoError(t, e)
	modelURL := string(modelRaw)
	old := openaiclient.OpenAIAPIBaseURL
	openaiclient.OpenAIAPIBaseURL = modelURL + "/v1/"
	defer func() { openaiclient.OpenAIAPIBaseURL = old }()
	oldAnthropicHost := anthropicclient.AnthropicAPIHost
	anthropicclient.AnthropicAPIHost = modelURL
	defer func() { anthropicclient.AnthropicAPIHost = oldAnthropicHost }()
	provider := os.Getenv("OV_FLOW_PROVIDER")
	if provider == "" {
		provider = "openai"
	}
	require.Contains(t, []string{"openai", "anthropic", "openai-compatible"}, provider)
	profile := os.Getenv("OV_FLOW_PROFILE")
	mode := os.Getenv("OV_FLOW_MODE")
	require.Contains(t, []string{"text-only", "tool-use"}, mode)
	withTools := mode == "tool-use"
	outputBytes, toolBytes, contextWindow, deltaMS := 2048, 4096, 272000, 20
	switch profile {
	case "medium":
		outputBytes, toolBytes, deltaMS = 16384, 65536, 5
	case "large":
		outputBytes, toolBytes, deltaMS = 49152, 524288, 5
	case "compaction":
		outputBytes, toolBytes, contextWindow, deltaMS = 49152, 65536, 64000, 5
	case "small", "mixed-paced":
	default:
		t.Fatalf("unknown profile %q", profile)
	}
	if !withTools {
		toolBytes = 0
	}
	agent := models.LLMConfig{Name: "App flow model", Provider: models.ProviderOpenAI, Model: "gpt-5.3-codex", AuthMethod: models.AuthMethodAPIKey, APIKey: "local-benchmark-only", MaxTokens: 65536, ContextWindow: contextWindow, ProviderContextWindow: contextWindow, CompactionThreshold: contextWindow * 3 / 4, IsDefault: true}
	transport := "responses-websocket"
	switch provider {
	case "anthropic":
		agent.Provider, agent.Model = models.ProviderAnthropic, "claude-opus-5"
		transport = "messages-sse"
	case "openai-compatible":
		agent.Provider, agent.Model = models.ProviderOpenAICompatible, "benchmark-compatible"
		agent.BaseURL = modelURL + "/v1"
		agent.CustomAuthConfigJSON = `{"allow_private_endpoints":true}`
		transport = "chat-completions-sse"
	}
	// Keep this growing-history comparison out of compaction. Both revisions
	// receive exactly the same explicit budget; compaction has a separate harness.
	if provider != "openai" {
		require.NotEqual(t, "compaction", profile)
		agent.ContextWindow, agent.ProviderContextWindow, agent.CompactionThreshold = 2000000, 2000000, 1500000
	}
	require.NoError(t, stores.LLMConfigs.Create(ctx, &agent))
	work := filepath.Join(dir, "repo")
	require.NoError(t, os.MkdirAll(work, 0700))
	project := models.Project{Name: "App flow", RepoPath: work}
	require.NoError(t, stores.Projects.Create(ctx, &project))
	manifest = appFlowManifest{Provider: provider, Transport: transport, Mode: mode, Project: project.ID, Model: agent.ID, OutputBytes: outputBytes, ToolBytes: toolBytes, Turns: 6}
	other := models.Task{ProjectID: project.ID, Title: "Other thread", Prompt: "Empty comparison thread", Category: models.CategoryCompleted, Status: models.StatusCompleted, AgentID: &agent.ID}
	require.NoError(t, stores.Tasks.Create(ctx, &other))
	manifest.OtherTask = other.ID
	workers, e := strconv.Atoi(os.Getenv("OV_FLOW_WORKERS"))
	require.NoError(t, e)
	require.Contains(t, []int{1, 10}, workers)
	tasks := make([]string, workers)
	rows := []map[string]any{}
	runtime.GC()
	runtime.GC() // clear startup pools; do NOT force GC between natural turns.
	var baseline runtime.MemStats
	runtime.ReadMemStats(&baseline)
	var basePages, baseSize int64
	require.NoError(t, conns.Reader.QueryRow("PRAGMA page_count").Scan(&basePages))
	require.NoError(t, conns.Reader.QueryRow("PRAGMA page_size").Scan(&baseSize))
	markPhase("setup-complete")
	for turn := 0; turn < manifest.Turns; turn++ {
		if profile == "mixed-paced" {
			outputBytes = []int{2048, 8192, 16384}[turn%3]
			toolBytes = []int{4096, 65536, 524288}[turn%3]
			deltaMS = 80
		}
		if !withTools {
			toolBytes = 0
		}
		toolFile := ""
		if withTools {
			toolFile = filepath.Join(work, fmt.Sprintf("evidence-%d.txt", turn))
			var data strings.Builder
			for n := 0; data.Len() < toolBytes; n++ {
				fmt.Fprintf(&data, "Record %05d for turn %d: distinct value=%d; inspect source path src/module_%d.go and preserve this evidence.\n", n, turn, n*37+turn, n%109)
			}
			fmt.Fprintf(&data, "TOOL_%d_END\n", turn)
			require.NoError(t, os.WriteFile(toolFile, []byte(data.String()), 0600))
		}
		control, _ := json.Marshal(map[string]any{"with_tools": withTools, "turn": turn, "output_bytes": outputBytes, "tool_file": toolFile, "delta_ms": deltaMS})
		resp, e := http.Post(modelURL+"/control", "application/json", strings.NewReader(string(control)))
		require.NoError(t, e)
		resp.Body.Close()
		// Diagnostic runs deliberately perturb GC/scheduling and must never be
		// included in resource comparisons. The ordinary workload is unchanged.
		diagnostic := diagnosticsEnabled && (turn == 0 || turn == 5)
		var traceFile *os.File
		if diagnostic {
			traceFile, e = os.Create(filepath.Join(dir, fmt.Sprintf("runtime-%d.trace", turn)))
			require.NoError(t, e)
			require.NoError(t, trace.Start(traceFile))
		}
		// Diagnostic snapshots synchronize GC accounting; timestamps describe the
		// perturbation explicitly. They are not lifetime measurements of objects.
		type allocationCapture struct {
			Name           string
			StartNS, EndNS int64
			Stats          runtime.MemStats
		}
		var allocationCaptures []allocationCapture
		captureAllocation := func(name string) {
			c := allocationCapture{Name: name, StartNS: time.Now().UnixNano()}
			runtime.GC()
			runtime.GC() // profile frees can lag by up to two GC cycles
			runtime.ReadMemStats(&c.Stats)
			f, err := os.Create(filepath.Join(dir, fmt.Sprintf("alloc-%d-%s.pprof", turn, name)))
			if err == nil {
				err = pprof.Lookup("allocs").WriteTo(f, 0)
				closeErr := f.Close()
				if err == nil {
					err = closeErr
				}
			}
			require.NoError(t, err)
			c.EndNS = time.Now().UnixNano()
			allocationCaptures = append(allocationCaptures, c)
		}
		profileAllocations := allocationDiagnostic && (turn == 0 || turn == 5)
		if profileAllocations {
			captureAllocation("before")
		}
		markPhase(fmt.Sprintf("turn-%d", turn))
		var before, after runtime.MemStats
		runtime.ReadMemStats(&before)
		cpu0, _ := appFlowCPU()
		start := time.Now()
		type sample struct {
			MS, CPU                        float64
			Heap, Alloc, Objects, Pause    uint64
			Frees, FreedBytes, LiveObjects uint64
			GC                             uint32
			RSS                            int64
		}
		samples := make([]sample, 1, 4096)
		samples[0].Heap = before.HeapAlloc
		samples[0].LiveObjects = before.HeapObjects
		makeSample := func(m runtime.MemStats, ms, cpu float64, rss int64) sample {
			return sample{MS: ms, CPU: cpu, Heap: m.HeapAlloc, Alloc: m.TotalAlloc - before.TotalAlloc,
				Objects: m.Mallocs - before.Mallocs, Pause: m.PauseTotalNs - before.PauseTotalNs,
				Frees: m.Frees - before.Frees, FreedBytes: (m.TotalAlloc - m.HeapAlloc) - (before.TotalAlloc - before.HeapAlloc),
				LiveObjects: m.HeapObjects, GC: m.NumGC - before.NumGC, RSS: rss}
		}
		stop, done := make(chan struct{}), make(chan struct{})
		go func() {
			defer close(done)
			captures := 0
			ticker := time.NewTicker(50 * time.Millisecond)
			defer ticker.Stop()
			for {
				select {
				case <-stop:
					return
				case <-ticker.C:
					var m runtime.MemStats
					runtime.ReadMemStats(&m)
					cpu, rss := appFlowCPU()
					samples = append(samples, makeSample(m, float64(time.Since(start).Microseconds())/1000, cpu-cpu0, rss))
					if profileAllocations && captures < 2 && time.Since(start) >= []time.Duration{time.Second, 3 * time.Second}[captures] {
						captureAllocation([]string{"stream-start", "stream-end"}[captures])
						captures++
					}
					if diagnostic && diagnosticMode == "1" && captures < 2 && time.Since(start) >= []time.Duration{100 * time.Millisecond, time.Second}[captures] {
						// Force GC only in labelled diagnostic runs so in-use samples
						// describe reachable memory rather than a previous GC cycle.
						runtime.GC()
						f, err := os.Create(filepath.Join(dir, fmt.Sprintf("live-%d-%d.pprof", turn, captures)))
						if err == nil {
							err = pprof.WriteHeapProfile(f)
							f.Close()
						}
						if err != nil {
							panic(err)
						}
						captures++
					}
				}
			}
		}()
		var wg sync.WaitGroup
		gate := make(chan struct{})
		for worker := 0; worker < workers; worker++ {
			wg.Add(1)
			go func(worker int) {
				defer wg.Done()
				<-gate
				prompt := fmt.Sprintf("Turn %d worker %d: explain the next findings and preserve prior conversation context.", turn, worker)
				if turn == 0 {
					task := models.Task{ProjectID: project.ID, Title: fmt.Sprintf("Concurrent flow %d", worker), Prompt: prompt, Category: models.CategoryActive, Status: models.StatusPending, AgentID: &agent.ID}
					require.NoError(t, stores.Tasks.Create(ctx, &task))
					tasks[worker] = task.ID
					execution, e := env.LLMSvc.ExecuteTaskWithAgent(ctx, task, agent)
					require.NoError(t, e)
					require.Equal(t, models.ExecCompleted, execution.Status)
				} else {
					form := url.Values{"message": {prompt}, "agent_id": {agent.ID}, "model_selection_managed": {"1"}}
					req := httptest.NewRequest(http.MethodPost, "/tasks/"+tasks[worker]+"/thread?project_id="+project.ID, strings.NewReader(form.Encode()))
					req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
					req.Header.Set("HX-Request", "true")
					rec := httptest.NewRecorder()
					env.Echo.ServeHTTP(rec, req)
					require.Equal(t, 200, rec.Code, rec.Body.String())
					var id string
					require.NoError(t, conns.Reader.QueryRow("SELECT id FROM executions WHERE task_id=? ORDER BY rowid DESC LIMIT 1", tasks[worker]).Scan(&id))
					subscribe(id)
				}
			}(worker)
		}
		close(gate)
		wg.Wait()
		deadline := time.Now().Add(2 * time.Minute)
		for env.Handler.updateWorkTracker.Active().Total() != 0 {
			require.True(t, time.Now().Before(deadline), "workers timed out")
			time.Sleep(10 * time.Millisecond)
		}
		elapsed := time.Since(start).Seconds() * 1000
		cpu1, rss := appFlowCPU()
		runtime.ReadMemStats(&after)
		close(stop)
		<-done
		markPhase(fmt.Sprintf("between-turns-%d", turn))
		if diagnostic {
			trace.Stop()
			require.NoError(t, traceFile.Close())
		}
		samples = append(samples, makeSample(after, elapsed, cpu1-cpu0, rss))
		if profileAllocations {
			captureAllocation("after")
			appFlowJSON(t, filepath.Join(dir, fmt.Sprintf("allocation-captures-%d.json", turn)), allocationCaptures)
		}
		peak := before.HeapAlloc
		for _, s := range samples {
			if s.Heap > peak {
				peak = s.Heap
			}
		}
		appFlowJSON(t, filepath.Join(dir, fmt.Sprintf("samples-%d.json", turn)), samples)
		if after.HeapAlloc > peak {
			peak = after.HeapAlloc
		}
		var count int
		require.NoError(t, conns.Reader.QueryRow("SELECT count(*) FROM executions WHERE status='completed'").Scan(&count))
		require.Equal(t, (turn+1)*workers, count)
		var pages, free, pageSize int64
		require.NoError(t, conns.Reader.QueryRow("PRAGMA page_count").Scan(&pages))
		require.NoError(t, conns.Reader.QueryRow("PRAGMA freelist_count").Scan(&free))
		require.NoError(t, conns.Reader.QueryRow("PRAGMA page_size").Scan(&pageSize))
		disk := map[string]int64{}
		for _, suffix := range []string{"", "-wal", "-shm"} {
			if s, e := os.Stat(filepath.Join(dir, "app.db") + suffix); e == nil {
				disk["db"+suffix] = s.Size()
			}
		}
		row := map[string]any{"provider": provider, "mode": mode, "profile": profile, "turn": turn, "assistant_target_bytes": outputBytes, "tool_file_target_bytes": toolBytes, "delta_ms": deltaMS, "wave_start_unix_ns": start.UnixNano(), "cpu_ms": cpu1 - cpu0, "elapsed_ms": elapsed, "alloc_bytes": after.TotalAlloc - before.TotalAlloc, "alloc_count": after.Mallocs - before.Mallocs, "free_count": after.Frees - before.Frees, "freed_bytes": (after.TotalAlloc - after.HeapAlloc) - (before.TotalAlloc - before.HeapAlloc), "start_live_objects": before.HeapObjects, "end_live_objects": after.HeapObjects, "start_heap": before.HeapAlloc, "gc_count": after.NumGC - before.NumGC, "gc_pause_ns": after.PauseTotalNs - before.PauseTotalNs, "sampled_peak_heap": peak, "end_heap": after.HeapAlloc, "process_peak_rss": rss, "db_pages_bytes": pages * pageSize, "db_free_bytes": free * pageSize, "files": disk}
		rows = append(rows, row)
		appFlowJSON(t, filepath.Join(dir, "turns.json"), rows)
		// Assertions and fixture IO are outside the measured turn. The next turn keeps
		// the normal live app state; no deliberate GC/reset between follow-ups.
	}
	markPhase("verification")
	for _, task := range tasks {
		history, e := stores.Executions.ListByTaskChronological(ctx, task)
		require.NoError(t, e)
		require.Len(t, history, manifest.Turns)
		for i, ex := range history {
			require.Contains(t, ex.Output, fmt.Sprintf("TURN_%d_END", i))
			if withTools {
				require.Contains(t, ex.Output, fmt.Sprintf("TOOL_%d_END", i))
			} else {
				require.NotContains(t, ex.Output, "TOOL_")
			}
		}
	}
	appFlowJSON(t, filepath.Join(dir, "turns.json"), rows)
	markPhase("final-gc")
	runtime.GC()
	runtime.GC()
	markPhase("final-validation")
	var retained runtime.MemStats
	runtime.ReadMemStats(&retained)
	appFlowJSON(t, filepath.Join(dir, "heap.json"), map[string]any{"baseline_heap": baseline.HeapAlloc, "retained_heap": retained.HeapAlloc, "retained_objects": retained.HeapObjects, "baseline_objects": baseline.HeapObjects, "initial_db_pages_bytes": basePages * baseSize})
	runtime.KeepAlive(env)
	runtime.KeepAlive(hub)
	appFlowJSON(t, filepath.Join(dir, "manifest.json"), manifest)
	resp, e := http.Get(modelURL + "/stats")
	require.NoError(t, e)
	defer resp.Body.Close()
	var peer map[string]any
	require.NoError(t, json.NewDecoder(resp.Body).Decode(&peer))
	appFlowJSON(t, filepath.Join(dir, "model.json"), peer)
	if profile == "compaction" {
		require.Greater(t, peer["compactions"].(float64), float64(0), "context never triggered compaction")
		// Summary fallback may depend on the profile; native continuation is mandatory.
	}
	require.Equal(t, float64(workers), peer["max_active"], "model streams did not overlap")
	requests := peer["requests"].([]any)
	if profile != "compaction" {
		require.Len(t, requests, workers*manifest.Turns*(1+map[bool]int{true: 1}[withTools]), "unexpected provider retries/rounds")
	}
	toolReplies := 0
	checkpointSeen := false
	for _, v := range requests {
		r := v.(map[string]any)
		require.Equal(t, provider, r["provider"], "wrong provider protocol")
		require.True(t, r["prior_turn_seen"] == true || r["checkpoint_seen"] == true, "prior conversation missing: %v", r)
		if profile != "compaction" {
			require.False(t, r["summary"] == true, "unexpected compaction/summary in concurrent comparison")
		}
		if r["tool_result_seen"] == true {
			toolReplies++
		}
		if r["checkpoint_seen"] == true {
			checkpointSeen = true
		}
	}
	if withTools {
		require.Equal(t, workers*manifest.Turns, toolReplies, "provider must receive each actual tool result")
	} else {
		require.Zero(t, toolReplies, "text baseline must not contain tool results")
	}
	if profile == "compaction" {
		require.True(t, checkpointSeen, "native checkpoint not restored")
	}
	broadcaster.Unsubscribe(eventSub)
	<-eventDone
	observers.Wait()
	require.Len(t, deliveries, workers*manifest.Turns)
	for _, d := range deliveries {
		require.Positive(t, d.Deltas)
		require.Positive(t, d.Persisted)
	}
	appFlowJSON(t, filepath.Join(dir, "delivery.json"), deliveries)
	markPhase("checkpoint")
	// Normalize storage only after the natural conversation has finished.
	_, e = conns.Writer.Exec("PRAGMA wal_checkpoint(TRUNCATE)")
	require.NoError(t, e)
	final, e := os.Stat(filepath.Join(dir, "app.db"))
	require.NoError(t, e)
	appFlowJSON(t, filepath.Join(dir, "storage.json"), map[string]any{"checkpointed_db_bytes": final.Size()})
}

// Phase markers are outside measured waves. Heap capture runs separately from
// timing: the external RSS sampler requests a capture on phase high-water rises.
// No GC is forced; Go's sampled in-use profile can lag by up to two GC cycles.
func appFlowMemoryPhases(t *testing.T, dir string) (func(string), func()) {
	type phase struct {
		Name   string
		UnixNS int64
	}
	phases := []phase{}
	mark := func(name string) {
		p := phase{name, time.Now().UnixNano()}
		phases = append(phases, p)
		raw, _ := json.Marshal(p)
		require.NoError(t, os.WriteFile(filepath.Join(dir, "phase.tmp"), raw, 0600))
		require.NoError(t, os.Rename(filepath.Join(dir, "phase.tmp"), filepath.Join(dir, "phase.json")))
	}
	stop, done := make(chan struct{}), make(chan struct{})
	enabled := os.Getenv("OV_FLOW_MEMORY_PROFILE") == "1"
	if enabled {
		appFlowJSON(t, filepath.Join(dir, "memory-diagnostic.json"), map[string]any{"forced_gc_at_peaks": false, "performance_comparison": false})
		go func() {
			defer close(done)
			ticker := time.NewTicker(25 * time.Millisecond)
			defer ticker.Stop()
			var previous int64
			var captures []map[string]any
			defer func() { appFlowJSON(t, filepath.Join(dir, "memory-captures.json"), captures) }()
			for {
				select {
				case <-stop:
					return
				case <-ticker.C:
					raw, err := os.ReadFile(filepath.Join(dir, "heap-request.json"))
					if err != nil {
						continue
					}
					var request struct {
						Phase  string
						UnixNS int64
						RSS    int64
					}
					if json.Unmarshal(raw, &request) != nil || request.UnixNS <= previous {
						continue
					}
					previous = request.UnixNS
					raw, _ = os.ReadFile(filepath.Join(dir, "phase.json"))
					var current phase
					_ = json.Unmarshal(raw, &current)
					var before, after runtime.MemStats
					runtime.ReadMemStats(&before)
					started := time.Now().UnixNano()
					name := fmt.Sprintf("memory-%s.pprof", request.Phase)
					file, e := os.Create(filepath.Join(dir, name))
					if e == nil {
						e = pprof.WriteHeapProfile(file)
						_ = file.Close()
					}
					runtime.ReadMemStats(&after)
					capture := map[string]any{"file": name, "requested_phase": request.Phase, "captured_phase": current.Name, "requested_unix_ns": request.UnixNS, "started_unix_ns": started, "ended_unix_ns": time.Now().UnixNano(), "requested_rss": request.RSS, "heap_before": before.HeapAlloc, "heap_after": after.HeapAlloc, "heap_sys": before.HeapSys, "heap_released": before.HeapReleased, "gc": before.NumGC}
					if e != nil {
						capture["error"] = e.Error()
						t.Errorf("memory capture: %v", e)
					}
					captures = append(captures, capture)
				}
			}
		}()
	} else {
		close(done)
	}
	return mark, func() { close(stop); <-done; appFlowJSON(t, filepath.Join(dir, "phases.json"), phases) }
}
